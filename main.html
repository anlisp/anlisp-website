// ============================================================ // THEME TOGGLE // Switches between "aero" (default) and "modern" [data-theme] // Remembers choice across every page via localStorage // ============================================================ (function () { const STORAGE_KEY = "site-theme"; const root = document.documentElement; function applyTheme(theme) { if (theme === "modern") { root.setAttribute("data-theme", "modern"); } else { root.removeAttribute("data-theme"); } updateButtonLabel(theme); } function updateButtonLabel(theme) { const btn = document.getElementById("themeToggle"); if (!btn) return; btn.textContent = theme === "modern" ? "✨ aero mode" : "◾ modern mode"; } function currentTheme() { return root.getAttribute("data-theme") === "modern" ? "modern" : "aero"; } function toggleTheme() { const next = currentTheme() === "modern" ? "aero" : "modern"; localStorage.setItem(STORAGE_KEY, next); applyTheme(next); } document.addEventListener("DOMContentLoaded", function () { // apply saved (or default) theme immediately const saved = localStorage.getItem(STORAGE_KEY) || "aero"; applyTheme(saved); // create the button if the page didn't already add one let btn = document.getElementById("themeToggle"); if (!btn) { btn = document.createElement("button"); btn.id = "themeToggle"; document.body.appendChild(btn); } btn.addEventListener("click", toggleTheme); }); })();
